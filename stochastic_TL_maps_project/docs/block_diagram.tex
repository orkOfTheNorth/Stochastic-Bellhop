%%═══════════════════════════════════════════════════════════════════════════════
%% ROW 4 — UQ Algorithm blocks
%%═══════════════════════════════════════════════════════════════════════════════
\node[algo] (mc) at (3.5,-9.0)
  {\mico\;\textbf{run\_mc.m}\\[1pt]
   \textit{Monte Carlo}\\[4pt]
   \footnotesize
   $\hat{\mathbb{E}}[TL]=\tfrac{1}{N}\sum_i TL_i$\\[2pt]
   $\hat{V}[TL]=\tfrac{1}{N\!-\!1}\sum_i(TL_i\!-\!\hat\mu)^2$\\[2pt]
   $P_\mathrm{detection}=P(TL<\mathrm{FOM})$\\[2pt]
   KDE:\;$\hat{P}\!=\!\tfrac{1}{N}\sum_i\Phi\!\bigl(\tfrac{\mathrm{FOM}-TL_i}{h_i}\bigr)$};

\node[algo, right=0.5cm of mc] (delta)
  {\mico\;\textbf{run\_delta.m}\\[1pt]
   \textit{Delta / FOSM}\\[4pt]
   \footnotesize
   $J_k=\partial TL/\partial\theta_k\;$(FD)\\[2pt]
   $\mathrm{Var}\approx\sum_k\sigma_k^2 J_k^2$\\[2pt]
   $P_\mathrm{no\,det}\geq1-\mathrm{Var}/(\mathrm{Var}+d^2)$};

\node[algo, right=0.5cm of delta] (pce)
  {\mico\;\textbf{run\_pce.m}\\[1pt]
   \textit{Polynomial Chaos}\\[4pt]
   \footnotesize
   $TL\approx\sum_{k=0}^K a_k\Psi_k(\xi)$\\[2pt]
   $a_k=\langle TL,\Psi_k\rangle/\langle\Psi_k^2\rangle$\\[2pt]
   LOO-CV:\;$\ell=\alpha\ell_\mathbb{E}+(1-\alpha)\ell_V$\\[2pt]
   $K^*=\arg\min_K\ell(K)$};

\node[algo, right=0.5cm of pce] (ln3)
  {\mico\;\textbf{run\_ln3.m}\\[1pt]
   \textit{LN3 — MOM (LHS) \& MLE (IID)}\\[4pt]
   \footnotesize
   $TL\!-\!\gamma\sim\mathrm{LN}(\mu,\sigma^2)$;\;$\gamma\!=\!0.95\min(TL)$\\[2pt]
   \textbf{LHS}\;\text{---}\;\textrm{MOM:}\;$\hat\sigma^2\!=\!\log(\hat M_2/\hat M_1^2)$\\[2pt]
   \textbf{IID}\;\text{---}\;\textrm{MLE:}\;$\hat\mu\!=\!\overline{\log Y},\;\hat\sigma\!=\!\mathrm{std}(\log Y)$\\[2pt]
   $\mathrm{RMSE}\!=\!\|F_\mathrm{MLE}\!-\!\hat{F}_\mathrm{emp}\|_2$};
