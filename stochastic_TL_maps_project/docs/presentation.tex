\begin{frame}{Motivation: Why Stochastic Transmission Loss?}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \textbf{Sonar planning requires TL}, but inputs are uncertain:
      \begin{itemize}
        \item Source frequency $f$ drifts with temperature
        \item Source depth $z_S$ fluctuates with platform motion
        \item Sound Velocity Profile (SVP) varies with ocean state
      \end{itemize}
      \vspace{0.8em}
      \textbf{Nominal deterministic TL} (single Bellhop run) gives
      no information about \alert{how much} TL can vary and
      \alert{how likely} a target is detectable.

      \vspace{0.8em}
      \textbf{This project:} compare four UQ methods for cost vs accuracy
      across 4 range/depth scenarios and 3 perturbation magnitudes.
    \end{column}
    \begin{column}{0.42\textwidth}
      \begin{block}{Key Question}
        Given $\theta \sim \mathcal{N}(\mu,\sigma^2)$,\\[3pt]
        what is $P(\TL(\theta) < \mathrm{FOM})$?\\[6pt]
        \textit{(Shadow probability / detection probability)}
      \end{block}
      \vspace{0.5em}
      \begin{alertblock}{Figure of Merit (FOM)}
        Detection threshold in dB.\\
        $P_{\text{detect}} = P(\TL < \mathrm{FOM})$
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Method 1: Monte Carlo ($N=1000$ LHS)}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Estimators:}
      \begin{align*}
        \hat\EX[\TL] &= \frac{1}{N}\sum_{i=1}^N TL_i \\[4pt]
        \hat\Var[\TL] &= \frac{1}{N-1}\sum_{i=1}^N (TL_i - \hat\mu)^2 \\[4pt]
        \hat P_{\text{det}} &= \frac{1}{N}\sum_{i=1}^N \mathbf{1}[TL_i < \mathrm{FOM}]
      \end{align*}

      \vspace{0.5em}
      \textbf{Sampling:} Latin Hypercube Sampling (LHS) ensures stratified
      coverage of the input space with exactly $N=1000$ samples.

      \vspace{0.5em}
      \textbf{Cost:} 1000 Bellhop runs per scenario/distribution/parameter.
      All other methods reuse this cache — \alert{zero marginal cost}.
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Convergence of} $\hat\EX$ \textbf{and} $\hat\Var$ vs.\ $N$
      (sub-samples of $N=1000$ cache):
      \begin{center}
        \includegraphics[width=\linewidth]{mc_conv_mean_svp.png}
      \end{center}
      {\small Mean converges by $N\approx50$;
       Var requires $N\approx200$--$500$.}
    \end{column}
  \end{columns}
\end{frame}

%% SLIDE 7 — Delta Theory
%% ──────────────────────────────────────────────────────────────────────────────
\begin{frame}{Method 2: Delta / FOSM (First-Order Second-Moment)}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \textbf{1st-order Taylor expansion} around nominal $\boldsymbol\mu$:
      \[
        TL(\boldsymbol\theta) \approx TL(\boldsymbol\mu)
          + \sum_k J_k(\theta_k - \mu_k)
      \]
      where $J_k = \partial TL/\partial\theta_k\big|_{\boldsymbol\mu}$
      (computed by finite differences — 3 extra Bellhop runs).

      \vspace{0.5em}
      \textbf{Variance propagation:}
      \[
        \Var[\TL] \approx \sum_k \sigma_k^2 J_k^2
      \]

      \textbf{Shadow probability} (Chebyshev upper bound):
      \[
        P_{\text{shd}} \geq 1 - \frac{\Var[\TL]}{\Var[\TL] + d^2}
      \]
      where $d = TL_{\text{nom}} - \mathrm{FOM}$.

      \vspace{0.5em}
      \textbf{Cost:} 3 FD evaluations per parameter.
    \end{column}
    \begin{column}{0.44\textwidth}
      \begin{alertblock}{When Delta works}
        $\sigma \leq 1\%$, flat bathymetry — TL is nearly linear.
        Var maps accurate; shadow prob reliable.
      \end{alertblock}
      \vspace{0.5em}
      \begin{alertblock}{When Delta fails}
        Slope environments, large $\sigma$ — shadow zone boundary
        is non-smooth; linearisation breaks.
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

%% ──────────────────────────────────────────────────────────────────────────────
%% SLIDE 9 — PCE Theory
%% ──────────────────────────────────────────────────────────────────────────────
\begin{frame}{Method 3: Polynomial Chaos Expansion (PCE)}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      Approximate TL as a polynomial in the standardised input
      $\xi = (\theta-\mu)/\sigma$:
      \[
        TL(\xi) \approx \sum_{k=0}^{K} a_k\,\Psi_k(\xi)
      \]
      where $\Psi_k$ are \textbf{Hermite polynomials}
      (orthogonal w.r.t.\ $\mathcal{N}(0,1)$).

      \vspace{0.6em}
      \textbf{Coefficients} via projection (reusing MC samples):
      \[
        a_k = \frac{\langle TL, \Psi_k\rangle}{\langle\Psi_k^2\rangle}
      \]

      \textbf{Then:} $\EX[\TL] = a_0$,
      $\Var[\TL] = \sum_{k=1}^{K} a_k^2 \langle\Psi_k^2\rangle$.

      \vspace{0.5em}
      \textbf{Order selection} $K^*$ via \alert{leave-one-out cross-validation
      (LOO-CV)} on the combined mean + variance error.
    \end{column}
    \begin{column}{0.44\textwidth}
      \begin{block}{LOO-CV Error (combined)}
        \[
          \ell(K) = \alpha\,\ell_{\mathbb{E}}(K) + (1-\alpha)\,\ell_V(K)
        \]
        $K^* = \arg\min_K \ell(K)$
      \end{block}
      \vspace{0.5em}
      \begin{block}{Why not just maximise $K$?}
        At $K\geq18$, the Vandermonde matrix becomes
        ill-conditioned despite $N/(K+1)\approx47$.
        High-order Hermite polynomials blow up at $|\xi|>2$,
        distorting the reconstructed distribution.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

%% SLIDE 13 — LN3 Theory
%% ──────────────────────────────────────────────────────────────────────────────
\begin{frame}{Method 4: 3-Parameter Lognormal (LN3) — MLE}
  \begin{columns}[T]
    \begin{column}{0.54\textwidth}
      \textbf{Model:} $TL - \gamma \sim \mathrm{LN}(\mu, \sigma^2)$

      \vspace{0.5em}
      \textbf{Shift parameter:} $\gamma = 0.95\min(TL)$ (heuristic)

      \vspace{0.5em}
      \textbf{MLE estimation} of $(\mu, \sigma)$ via MATLAB \texttt{lognfit}
      on shifted samples $Y = TL - \gamma$:
      \begin{align*}
        \hat\mu &= \overline{\log Y} \\
        \hat\sigma &= \mathrm{std}(\log Y)
      \end{align*}

      \vspace{0.5em}
      \textbf{Shadow probability:}
      \[
        P(TL < \mathrm{FOM}) = \Phi_\mathrm{LN}(\mathrm{FOM}-\gamma;\,\hat\mu,\hat\sigma)
      \]

      \vspace{0.5em}
      \textbf{Validation:} KS statistic $D = \sup_x |F_\mathrm{LN}(x) - \hat F_\mathrm{emp}(x)|$
    \end{column}
    \begin{column}{0.42\textwidth}
      \begin{block}{MLE vs MOM}
        Method of moments (MOM) estimator:
        \begin{align*}
          \hat\sigma^2_\mathrm{MOM} &= \log(\hat M_2/\hat M_1^2) \\
          \hat\mu_\mathrm{MOM} &= \log(\hat M_1) - \tfrac{1}{2}\hat\sigma^2
        \end{align*}
        \alert{MLE outperforms MOM} at nearly all pixels
        (lower KS statistic), especially at small sample sizes
        and near-symmetric distributions.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

%% ──────────────────────────────────────────────────────────────────────────────
%% SLIDE 18 — SVP Perturbation
%% ──────────────────────────────────────────────────────────────────────────────
\begin{frame}{SVP Perturbation: Physical Interpretation}
  \begin{columns}[T]
    \begin{column}{0.54\textwidth}
      The SVP perturbation model applies a uniform temperature shift
      $\Delta T \sim \mathcal{N}(0,\sigma_s^2)$ to the entire profile,
      coupled with a mixed-layer depth (MLD) change:
      \begin{itemize}
        \item Warmer $\Delta T > 0$: steeper gradient, stronger downward refraction
        \item Cooler $\Delta T < 0$: weaker gradient, surface duct collapses
      \end{itemize}
      \vspace{0.5em}
      This creates \alert{non-linear} shifts in shadow zone boundaries —
      particularly sensitive in upslope/downslope scenarios.
    \end{column}
    \begin{column}{0.42\textwidth}
      \begin{block}{SVP uncertainty effect}
        SVP perturbation at 10\% causes the largest variance maps
        among the three parameters in deep water,
        because the convergence zone boundary is highly sensitive
        to sound speed gradient changes.
      \end{block}
      \vspace{0.4em}
      \begin{alertblock}{PCE for SVP}
        $K^*$ for SVP: 5--8 (flat), consistent convergence.
        SVP does \textbf{not} show the $z_S$-type resistance
        that prevents convergence in deep water.
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

%% ──────────────────────────────────────────────────────────────────────────────
%% SLIDE 19 — Key Findings

\begin{frame}{Key Findings}
  \begin{enumerate}\itemsep10pt
    \item \textbf{Delta is near-exact at 1\% perturbation} for flat bathymetry.
          Only 3 extra Bellhop runs needed.
          Fails at slope boundaries where TL is non-smooth in parameter space.

    \item \textbf{PCE converges with $K^*=5$--$9$} for flat scenarios (freq, SVP).
          Slope environments reach $K^*=10$--$12$, signalling that polynomial
          surrogacy approaches its limit.
          $z_S$ uncertainty in deep water \alert{resists PCE convergence} even at
          $K_{\max}=20$: the shadow zone shifts non-linearly with source depth.

    \item \textbf{LN3-MLE is reliable} (KS $<\!0.05$) across most of the
          range--depth grid.
          Fails at bimodal interference nodes; use empirical MC or Chebyshev there.

    \item \textbf{All analytical methods reuse the $N=1000$ MC cache} at zero
          marginal Bellhop cost.
          The question is not ``can we afford PCE?'' but
          ``\alert{does the PCE surrogate add information beyond empirical MC?}''
          For flat bathymetry at $\sigma=5$--$10\%$: \textbf{yes}.
  \end{enumerate}
\end{frame}

\begin{frame}{Conclusions and Future Work}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \textbf{Conclusions:}
      \begin{itemize}\itemsep5pt
        \item Four UQ methods benchmarked across 36 scenario/distribution combos
        \item Delta is the cheapest and best for small perturbations
        \item PCE extends reliable UQ to moderate perturbations at zero extra cost
        \item LN3-MLE provides smooth, compact shadow-probability maps
        \item MC remains necessary for slope environments and $z_S$ uncertainty
        \item Detection probability can be bounded tightly with Chebyshev
              or estimated compactly with LN3-MLE
      \end{itemize}
    \end{column}
